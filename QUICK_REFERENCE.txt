 常用命令速查

# 1. 训练baseline (7通道)
python scripts/training/train_7ch.py \
  --data configs/datasets/marida_mask_rescue_7ch.yaml \
  --epochs 30 --batch 8 --imgsz 512

# 2. 训练10通道光谱增强
python scripts/training/train_10ch_spectral.py \
  --data configs/datasets/marida_mask_rescue_10ch.yaml \
  --epochs 30 --batch 8

# 3. 数据预处理
python scripts/preprocessing/preprocess_extract_bands.py \
  --bands 10 --normalize

# 4. 验证模型
python scripts/evaluation/validate_annotations.py \
  --weights experiments/runs/runs_marida/xxx/weights/best.pt

# 5. 可视化结果
python scripts/visualization/visualize_dataset.py \
  --data data/processed/mask_bbox_rescue_10ch
